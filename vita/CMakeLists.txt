cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain")
endif()
project(nbbatman_vita C CXX)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(VITA_APP_NAME "Ninja Baseball Bat Man")
set(VITA_TITLEID  "NBBM00001")
set(VITA_VERSION  "01.00")
set(CMAKE_CXX_STANDARD 17)
set(R ${CMAKE_SOURCE_DIR}/..)
get_filename_component(ROOT_ABS ${R} ABSOLUTE)
# -ffile-prefix-map: no build folder in the binary, so a build is the same byte for byte from any folder
set(COMMON_FLAGS "-O2 -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard -fno-strict-aliasing -ffile-prefix-map=${ROOT_ABS}=.")
# -DNBB_PROFILE=ON: what psp2-tooling needs (DWARF, a fixed base); ROMs, log and dumps on host0:. Same -O2 code.
option(NBB_PROFILE "symbols + no ASLR + run from host0: for psp2-tooling" OFF)
if(NBB_PROFILE)
  set(COMMON_FLAGS "${COMMON_FLAGS} -g -fno-omit-frame-pointer")
  set(NBB_SELF_FLAGS NOASLR)
endif()
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${COMMON_FLAGS}")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${COMMON_FLAGS}")

file(GLOB V33_GEN ${R}/gen/v33/*.cpp)
add_library(nbb_gen STATIC ${V33_GEN})
target_include_directories(nbb_gen PUBLIC ${R}/host)
target_compile_options(nbb_gen PRIVATE -Wno-unused-variable -Wno-unused-label)

# YM2151 FM synthesis: ymfm (Aaron Giles, BSD-3-Clause)
set(YMFM ${R}/third_party/ymfm/src)
add_library(ymfm STATIC ${YMFM}/ymfm_opm.cpp)
target_include_directories(ymfm PUBLIC ${YMFM})

add_executable(nbbatman ${R}/host/v33.cpp ${R}/host/pic.cpp ${R}/host/board.cpp ${R}/host/video.cpp ${R}/host/rec.cpp
               ${R}/host/machine.cpp ${R}/host/sound.cpp ${R}/host/png.cpp ${CMAKE_SOURCE_DIR}/vita_main.cpp)
target_include_directories(nbbatman PRIVATE ${R}/host)
target_compile_definitions(nbbatman PRIVATE NBB_YMFM)
if(NBB_PROFILE)
  target_compile_definitions(nbbatman PRIVATE NBB_DUMP)
endif()
target_compile_options(nbbatman PRIVATE -include ${CMAKE_SOURCE_DIR}/vita_log.h)
target_link_libraries(nbbatman nbb_gen ymfm vitaGL vitashark SceShaccCgExt mathneon taihen_stub SceShaccCg_stub
    SceGxm_stub SceDisplay_stub SceCtrl_stub SceSysmodule_stub SceKernelDmacMgr_stub SceAppMgr_stub
    SceAppUtil_stub SceCommonDialog_stub SceIofilemgr_stub ScePower_stub SceLibKernel_stub SceKernelThreadMgr_stub
    SceRtc_stub SceAudio_stub SceTouch_stub SceMotion_stub m stdc++)
vita_create_self(eboot.bin nbbatman UNSAFE ${NBB_SELF_FLAGS})
# LiveArea images are not in the repository: packed when present (8-bit indexed PNGs)
set(VPK_FILES FILE ${CMAKE_SOURCE_DIR}/sce_sys/livearea/contents/template.xml sce_sys/livearea/contents/template.xml)
foreach(ART icon0.png livearea/contents/bg.png livearea/contents/startup.png)
  if(EXISTS ${CMAKE_SOURCE_DIR}/sce_sys/${ART})
    list(APPEND VPK_FILES FILE ${CMAKE_SOURCE_DIR}/sce_sys/${ART} sce_sys/${ART})
  endif()
endforeach()
vita_create_vpk(nbbatman.vpk ${VITA_TITLEID} eboot.bin VERSION ${VITA_VERSION} NAME ${VITA_APP_NAME} ${VPK_FILES})
